    <script>
        // Configuration
        // Season 2 was announced around Oct 8, 2023.
        const START_DATE = new Date('2023-10-08T00:00:00').getTime();
        // Estimated Release: April 1, 2026 (Placeholder for April 2026 window)
        const TARGET_DATE = new Date('2026-04-01T00:00:00').getTime();

        const elMonths = document.getElementById('months');
        const elDays = document.getElementById('days');
        const elMinutes = document.getElementById('minutes');
        const elSeconds = document.getElementById('seconds');
        const elProgressBar = document.getElementById('progress-bar');
        const elPercentage = document.getElementById('percentage-text');

        function updateCountdown() {
            const now = new Date().getTime();
            const distance = TARGET_DATE - now;

            if (distance < 0) {
                // If we passed the date
                elMonths.innerText = "0";
                elDays.innerText = "0";
                elMinutes.innerText = "0";
                elSeconds.innerText = "0";
                elProgressBar.style.width = "100%";
                elPercentage.innerText = "100% (Arrived!)";
                return;
            }

            // --- Time Calculations ---
            
            // 1. Calculate Months
            // We'll use a precise Date object approach to get remaining months
            let startDateForCalc = new Date(now);
            let targetDateForCalc = new Date(TARGET_DATE);
            let months = (targetDateForCalc.getFullYear() - startDateForCalc.getFullYear()) * 12;
            months -= startDateForCalc.getMonth();
            months += targetDateForCalc.getMonth();
            
            // If we are currently on a day later in the month than the target day (1st), subtract a month
            // Example: Now is Jan 20, Target is Feb 1. Difference is 1 month, but we haven't reached the 20th of Feb yet.
            // Actually simpler: Subtract one month if the current day > target day (1)
            // But since Target is fixed at day 1, and 'now' is likely > 1, we usually subtract 1 month
            // unless 'now' is also the 1st.
            if (startDateForCalc.getDate() > targetDateForCalc.getDate()) {
                months--;
            }

            // 2. Calculate remaining duration after removing the full months
            // Move start date forward by 'months'
            let tempDate = new Date(now);
            tempDate.setMonth(tempDate.getMonth() + months);
            
            let remainingTimeAfterMonths = TARGET_DATE - tempDate.getTime();

            // 3. Days
            const days = Math.floor(remainingTimeAfterMonths / (1000 * 60 * 60 * 24));
            
            // 4. Hours (Internal calculation only, to add to minutes as per user request)
            const hoursRemaining = Math.floor((remainingTimeAfterMonths % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            
            // 5. Minutes
            // User asked for "Months, Days, Minutes, Seconds". 
            // So we combine Hours into Minutes: (Hours * 60) + Standard Minutes
            const standardMinutes = Math.floor((remainingTimeAfterMonths % (1000 * 60 * 60)) / (1000 * 60));
            const totalMinutes = (hoursRemaining * 60) + standardMinutes;

            // 6. Seconds
            const seconds = Math.floor((remainingTimeAfterMonths % (1000 * 60)) / 1000);

            // Update DOM
            elMonths.innerText = months;
            elDays.innerText = days;
            elMinutes.innerText = totalMinutes; // Showing total minutes including hours
            elSeconds.innerText = seconds;

            // --- Progress Bar Calculations ---
            const totalDuration = TARGET_DATE - START_DATE;
            const timeElapsed = now - START_DATE;
            let percentage = (timeElapsed / totalDuration) * 100;
            
            // Clamp percentage between 0 and 100
            percentage = Math.max(0, Math.min(100, percentage));

            elProgressBar.style.width = percentage + "%";
            elPercentage.innerText = percentage.toFixed(4) + "%";
        }

        // Feather Animation Spawner
        function createFeather() {
            const container = document.getElementById('feather-container');
            const feather = document.createElement('div');
            feather.classList.add('feather');
            
            // Randomize position and animation duration
            feather.style.left = Math.random() * 100 + 'vw';
            feather.style.animationDuration = (Math.random() * 5 + 5) + 's'; // 5-10s
            feather.style.opacity = Math.random() * 0.5;
            
            container.appendChild(feather);

            // Remove after animation
            setTimeout(() => {
                feather.remove();
            }, 10000);
        }

        // Init
        updateCountdown();
        setInterval(updateCountdown, 1000);
        setInterval(createFeather, 800);

    </script>
